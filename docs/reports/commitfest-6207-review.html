Patch summary

The patch prevents Windows socket handle inheritance to child processes:

  1. pgwin32_socket() now passes WSA_FLAG_NO_HANDLE_INHERIT to WSASocket() — newly created sockets (including the listening socket) are non-inheritable.
  2. BackendInitialize() calls SetHandleInformation(... HANDLE_FLAG_INHERIT, 0) on the client socket under #ifdef WIN32 — clears inheritance on the backend's client socket (which read_inheritable_socket recreates as inheritable via WSASocket(FROM_PROTOCOL_INFO,...,0) with dwFlags=0).
  3. New TAP test src/bin/pg_ctl/t/005_socket_handle_inheritance.pl (Windows-only).

Core fix correctness: The two-pronged approach (creation-time flag + post-inheritance clear) is correct for the stated zombie-listening-port bug. The approach is endorsed on-list by Thomas Munro (thread 626963, 2025-11-07). No catalog/ABI change; no catversion bump needed.

+# Test that socket handles are not inherited by child processes on Windows.
+#
+# Without the fix, a detached grandchild spawned via COPY TO PROGRAM
+# inherits the listening socket; the kernel's reference counting keeps the
+# port bound after the postmaster dies, and a restart on the same port fails.
+# With the fix, the listening socket is not inherited and the restart succeeds.
+
+use strict;
+use warnings FATAL => 'all';
+
+use PostgreSQL::Test::Cluster;
+use PostgreSQL::Test::Utils;
+use Test::More;
+use Time::HiRes qw(sleep);
+
+plan skip_all => 'this test exercises Windows socket handle inheritance'
+  if $^O ne 'MSWin32';
+
+my $node = PostgreSQL::Test::Cluster->new('main');
+$node->init;
+$node->start;
+
+my $port = $node->port;
+my $data_dir = $node->data_dir;
+my $marker_file = "$data_dir/copy_marker.txt";
+my $pidfile     = "$data_dir/copy_grandchild_pid.txt";
+unlink $marker_file, $pidfile;
+
+my $copy_cmd =
+    q{powershell -NoProfile -Command "}
+  . qq{Set-Content -Path '$marker_file' -Value 'started' -Encoding ASCII; }
+  . qq{\$p = Start-Process -FilePath powershell -ArgumentList '-NoProfile','-Command','Start-Sleep 300' -WindowStyle Hidden -PassThru; }
+  . qq{\$p.Id | Out-File -FilePath '$pidfile' -Encoding ASCII; }
+  . q{exit 0"}
+  . q{};
+
+$node->safe_psql(
+	'postgres',
+	qq{COPY (SELECT 1) TO PROGRAM '$copy_cmd';});
+
+my $grandchild_pid;
+ok( PostgreSQL::Test::Utils::poll_execute(
+		sub { -e $marker_file && -e $pidfile; }
+	),
+	'detached grandchild spawned by backend COPY TO PROGRAM'
+);
+
+{
+	open my $fh, '<', $pidfile or die "could not open $pidfile: $!";
+	$grandchild_pid = <$fh>;
+	close $fh;
+	chomp $grandchild_pid if defined $grandchild_pid;
+}
+diag("grandchild pid: $grandchild_pid") if defined $grandchild_pid;
+
+$node->kill9;
+sleep 0.5;
+
+my $restarted = $node->start(fail_ok => 1);
+ok($restarted, "postmaster restarted on port $port after crash with detached grandchild running");
+
+if (!$restarted)
+{
+	diag("restart failed - listening port was not freed; ");
+	diag("this indicates the detached grandchild inherited the listening socket");
+}
+
+$node->stop('immediate');
+if (defined $grandchild_pid && $grandchild_pid =~ /^\d+$/)
+{
+	system(
+		qq{powershell -NoProfile -Command "Stop-Process -Id $grandchild_pid -Force -ErrorAction SilentlyContinue" 2>$null}
+	);
+}
+unlink $marker_file, $pidfile;
+
+done_testing();
